sudo nano /etc/systemd/system/django_plc.service

[Unit]
Description=Django PLC Server
After=network.target

[Service]
User=admin
Group=admin
WorkingDirectory=/home/admin/plc_250_main
Environment="PATH=/home/admin/plc_250_main/env/bin"
ExecStart=/home/admin/plc_250_main/env/bin/python manage.py runserver 0.0.0.0:6013

RestartSec=5

[Install]
WantedBy=multi-user.target
